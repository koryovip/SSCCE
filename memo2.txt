DECLARE
    l_table_name    VARCHAR2(128);
    l_sql           CLOB;
    l_col_list      CLOB;
    l_val_list      CLOB;
    l_col_count     NUMBER := 0;
    l_csv_col       VARCHAR2(10); -- col001 などの文字列を一時格納
BEGIN
    -- 1. テーブル名の検証
    l_table_name := DBMS_ASSERT.SQL_OBJECT_NAME(UPPER(:table_name));

    -- 2. INSERT文の基盤作成
    l_sql := 'INSERT INTO ' || l_table_name || ' (';

    -- 3. 列情報とデータ型を取得し、マッピングと型変換を自動生成
    FOR rec IN (
        SELECT column_name, column_id, data_type 
          FROM user_tab_columns
         WHERE table_name = l_table_name
         ORDER BY column_id
    ) LOOP
        l_col_count := l_col_count + 1;

        IF l_col_count > 1 THEN
            l_col_list := l_col_list || ', ';
            l_val_list := l_val_list || ', ';
        END IF;

        -- 実際のテーブルの列名を追加
        l_col_list := l_col_list || rec.column_name;
        
        -- CSVパーサー側の列名 (col001, col002, ...) を生成
        l_csv_col := 'col' || LPAD(l_col_count, 3, '0');

        -- ★ここから追加: データ型に応じた明示的な型変換★
        IF rec.data_type = 'DATE' THEN
            -- CSVの日付フォーマットが 'YYYY/MM/DD' であると想定
            l_val_list := l_val_list || 'TO_DATE(' || l_csv_col || ', ''YYYY/MM/DD'')';

        ELSIF rec.data_type LIKE 'TIMESTAMP%' THEN
            -- タイムスタンプ型の場合
            l_val_list := l_val_list || 'TO_TIMESTAMP(' || l_csv_col || ', ''YYYY/MM/DD HH24:MI:SS'')';

        ELSIF rec.data_type IN ('NUMBER', 'FLOAT') THEN
            -- 数値型の場合 (カンマ区切りの数値を除去したい場合は書式モデルを追加)
            l_val_list := l_val_list || 'TO_NUMBER(' || l_csv_col || ')';

        ELSE
            -- VARCHAR2, CHAR などの文字列型はそのまま
            l_val_list := l_val_list || l_csv_col;
        END IF;

    END LOOP;

    IF l_col_count = 0 THEN
        raise_application_error(-20001, 'テーブルが見つからないか、アクセス権限がありません。');
    END IF;

    -- 4. 動的SQLの組み立て
    l_sql := l_sql || l_col_list || ') SELECT ' || l_val_list ||
             ' FROM TABLE(apex_data_parser.parse(' ||
             ' p_content => :1, ' ||       
             ' p_file_name => ''import.csv'', ' ||
             ' p_skip_rows => 1))';        

    -- 5. 動的SQLの実行
    EXECUTE IMMEDIATE l_sql USING :body;

    :status := 201;

EXCEPTION
    WHEN OTHERS THEN
        :status := 400;
        -- 開発用デバッグ出力
        HTP.p('{"error": "' || SQLERRM || '"}');
END;
