services:
  db:
    image: mariadb:11.8
    restart: always
    command: --transaction-isolation=READ-COMMITTED --log-bin=binlog --binlog-format=ROW
    environment:
      MYSQL_ROOT_PASSWORD: 改成root密码
      MYSQL_DATABASE: nextcloud
      MYSQL_USER: nextcloud
      MYSQL_PASSWORD: 改成数据库密码
    volumes:
      - ./data/db:/var/lib/mysql

  nextcloud:
    image: nextcloud:34   # 部署前到 Docker Hub 确认当前最新大版本
    restart: always
    depends_on:
      - db
    ports:
      - "8081:80"
    environment:
      MYSQL_HOST: db
      MYSQL_DATABASE: nextcloud
      MYSQL_USER: nextcloud
      MYSQL_PASSWORD: 改成数据库密码   # 与上面保持一致
      NEXTCLOUD_ADMIN_USER: admin
      NEXTCLOUD_ADMIN_PASSWORD: 改成管理员密码
      NEXTCLOUD_TRUSTED_DOMAINS: 服务器IP   # 例: 192.168.1.10
    volumes:
      - ./data/nextcloud:/var/www/html

  onlyoffice:
    image: onlyoffice/documentserver:9.4.0
    restart: always
    ports:
      - "8080:80"
    environment:
      JWT_ENABLED: "true"
      JWT_SECRET: 改成你的密钥
    volumes:
      - ./data/onlyoffice/logs:/var/log/onlyoffice
      - ./data/onlyoffice/data:/var/www/onlyoffice/Data
      - ./data/onlyoffice/lib:/var/lib/onlyoffice
      - ./data/onlyoffice/db:/var/lib/postgresql
